   197→   197→  startRun(opts: { fast?: boolean } = {}): void {

   203→   203→  /** Следующий уровень забега (счёт сохраняется) */
   204→   204→  startLevel(level: number, perk: PerkId | null): void {
   205→   205→    this.lvl = level
   206→   206→    this.perk = perk
   207→   207→    this.duration = perk === 'timeplus' ? LEVEL_SECONDS_PERK : LEVEL_SECONDS
   208→   208→    this.quota = this.debugFast ? 3 : quotaFor(level, perk === 'doublepoints')
   209→   209→    this.collected = 0
   210→   210→    this.wave = 1
   211→   211→    this.waveSize = Math.min(16, 7 + Math.floor(level * 0.8))
   212→   212→    this.cornChance = Math.min(0.3, 0.12 + (level - 1) * 0.02)
   213→   213→    if (perk === 'cornrain') this.cornChance = Math.min(0.5, this.cornChance * 2)
   214→   214→    this.goldenChance = perk === 'goldenvein' ? 0.26 : 0.12
   215→   215→    this.baseWaveDelay = Math.max(0.35, 0.82 - (level - 1) * 0.04)
   216→   216→
   217→   217→    this.phase = 'playing'
   218→   218→    this.paused = false
   219→   219→    this.timeLeft = this.duration
   220→   220→    this.pendingClear = -1
   221→   221→    this.pieces = []
   222→   222→    this.carried = []
   223→   223→    this.flying = []
   224→   224→    this.particles = []
   225→   225→    this.texts = []
   226→   226→    this.waveDelay = -1
   227→   227→    this.lastSec = Math.ceil(this.timeLeft)
   228→   228→    this.banner = { text: `УРОВЕНЬ ${level}`, t: 0 }
   229→   229→    audio.setIntensity(0)
   230→   230→    audio.setMusicLevel(level >= 3 ? 1 : 0)

   252→   252→    if (this.phase !== 'playing' || this.paused) return
   253→   253→    this.paw.tx = x

   260→   260→  /** Размер игрового поля в CSS-пикселях (задаёт компонент с учётом перков) */
   261→   261→  handleResize(w: number, h: number): void {
   262→   262→    if (w === 0 || h === 0) return
   263→   263→
   264→   264→    const oldW = this.W
   265→   265→    const oldH = this.H
   266→   266→    this.dpr = Math.min(window.devicePixelRatio || 1, 2)
   267→   267→    this.W = w
   268→   268→    this.H = h
   269→   269→    this.canvas.width = Math.round(w * this.dpr)
   270→   270→    this.canvas.height = Math.round(h * this.dpr)
   271→   271→    this.ctx.setTransform(this.dpr, 0, 0, this.dpr, 0, 0)
   272→   272→    this.ctx.imageSmoothingEnabled = false
   273→   273→
   274→   274→    this.u = Math.max(0.7, Math.min(1.5, Math.min(w, h) / 420))
   275→   275→
   276→   276→    if (oldW > 0 && oldH > 0 && (oldW !== w || oldH !== h)) {
   277→   277→      const kx = w / oldW
   278→   278→      const ky = h / oldH
   279→   279→      for (const p of this.pieces) {
   280→   280→        p.x *= kx
   281→   281→        p.y *= ky
   282→   282→      }
   283→   283→      this.paw.x *= kx
   284→   284→      this.paw.y *= ky
   285→   285→      this.paw.tx *= kx
   286→   286→      this.paw.ty *= ky
   287→   287→    }
   288→   288→
   289→   289→    this.cupScale = 4.6 * this.u
   290→   290→    this.cupH = 26 * this.cupScale
   291→   291→    this.cup.x = w / 2
   292→   292→    this.cup.y = h - this.cupH / 2 - 12 * this.u
   293→   293→  }

   295→   295→  /* ================= волны ================= */
   296→   296→
   297→   297→  private spawnWave(): void {
   298→   298→    let cornOnField = this.pieces.filter((p) => p.kind === 'corn').length
   299→   299→    for (let i = 0; i < this.waveSize; i++) {
   300→   300→      const kind = this.decideKind(cornOnField)
   301→   301→      if (kind === 'corn') cornOnField++
   302→   302→      this.spawnPiece(i, kind)
   303→   303→    }
   304→   304→  }
   305→   305→
   306→   306→  private decideKind(cornOnField: number): PieceKind {
   307→   307→    const r = Math.random()
   308→   308→    if (r < this.cornChance && cornOnField < CORN_MAX) return 'corn'
   309→   309→    return Math.random() < this.goldenChance ? 'golden' : 'popcorn'
   310→   310→  }
   311→   311→
   312→   312→  private spawnPiece(index: number, kind: PieceKind): void {
   313→   313→    const u = this.u
   314→   314→    const mX = 46 * u
   315→   315→    const top = 80 + 16 * u
   316→   316→    const bottom = this.H - 46 * u
   317→   317→    let x = this.W / 2
   318→   318→    let y = this.H / 2
   319→   319→    for (let tries = 0; tries < 40; tries++) {
   320→   320→      x = mX + Math.random() * (this.W - mX * 2)
   321→   321→      y = top + Math.random() * Math.max(10, bottom - top)
   322→   322→      const dc = Math.hypot(x - this.cup.x, y - this.cup.y)
   323→   323→      if (dc <= 135 * u) continue
   324→   324→      // не даём зёрнам застрять в кластере — маршрут всегда проходим
   325→   325→      let crowded = false
   326→   326→      for (const q of this.pieces) {
   327→   327→        if (Math.hypot(x - q.x, y - q.y) < 40 * u) {
   328→   328→          crowded = true
   329→   329→          break
   330→   330→        }
   331→   331→      }
   332→   332→      if (!crowded) break
   333→   333→    }
   334→   334→    this.pieces.push({
   335→   335→      x,
   336→   336→      y,
   337→   337→      rot: (Math.random() - 0.5) * 0.7,
   338→   338→      kind,
   339→   339→      phase: Math.random() * Math.PI * 2,
   340→   340→      spawn: SPAWN_DUR + index * 0.045,
   341→   341→      dA: Math.random() * Math.PI * 2,
   342→   342→    })
   343→   343→  }

   345→   345→  /* ================= перки-механики ================= */
   346→   346→
   347→   347→  /** «Скользкий пол»: зёрна катаются по полу */
   348→   348→  private driftPieces(dt: number): void {
   349→   349→    const u = this.u
   350→   350→    const sp = 17 * u
   351→   351→    const mX = 34 * u
   352→   352→    const top = 90
   353→   353→    const bottom = this.H - 42 * u
   354→   354→    for (const p of this.pieces) {
   355→   355→      p.dA += (Math.random() - 0.5) * dt * 2.4
   356→   356→      p.x += Math.cos(p.dA) * sp * dt
   357→   357→      p.y += Math.sin(p.dA) * sp * dt
   358→   358→      if (p.x < mX) {
   359→   359→        p.x = mX
   360→   360→        p.dA = Math.PI - p.dA
   361→   361→      }
   362→   362→      if (p.x > this.W - mX) {
   363→   363→        p.x = this.W - mX
   364→   364→        p.dA = Math.PI - p.dA
   365→   365→      }
   366→   366→      if (p.y < top) {
   367→   367→        p.y = top
   368→   368→        p.dA = -p.dA
   369→   369→      }
   370→   370→      if (p.y > bottom) {
   371→   371→        p.y = bottom
   372→   372→        p.dA = -p.dA
   373→   373→      }
   374→   374→    }
   375→   375→  }
   376→   376→
   377→   377→  /** «Магнитная лапа»: попкорн сам тянется к лапке */
   378→   378→  private magnetPull(dt: number): void {
   379→   379→    const u = this.u
   380→   380→    const r = 115 * u
   381→   381→    const k = Math.min(1, dt * 3.4)
   382→   382→    for (const p of this.pieces) {
   383→   383→      if (p.kind === 'corn' || p.spawn > 0.12) continue
   384→   384→      const d = Math.hypot(this.paw.x - p.x, this.paw.y - p.y)
   385→   385→      if (d < r && d > 1) {
   386→   386→        p.x += (this.paw.x - p.x) * k
   387→   387→        p.y += (this.paw.y - p.y) * k
   388→   388→      }
   389→   389→    }
   390→   390→  }
   391→   391→
   392→   392→  /* ================= обновление ================= */
   393→   393→
   394→   394→  private update(dt: number): void {
   395→   395→    const u = this.u
   396→   396→
   397→   397→    // Плавное следование лапки за пальцем/курсором («Тяжёлая лапа» — вдвое медленнее)
   398→   398→    if (!this.paw.seen) {
   399→   399→      this.paw.tx = this.cup.x + Math.sin(performance.now() * 0.001) * 80 * u
   400→   400→      this.paw.ty = this.H * 0.35
   401→   401→    }
   402→   402→    const k = 1 - Math.exp(-dt * (this.perk === 'slowpaw' ? 6.5 : 14))
   403→   403→    this.paw.x += (this.paw.tx - this.paw.x) * k
   404→   404→    this.paw.y += (this.paw.ty - this.paw.y) * k
   405→   405→    this.paw.x = Math.max(22 * u, Math.min(this.W - 22 * u, this.paw.x))
   406→   406→    this.paw.y = Math.max(70, Math.min(this.H - 20 * u, this.paw.y))
   407→   407→
   408→   408→    if (this.perk === 'drift') this.driftPieces(dt)
   409→   409→    if (this.perk === 'magnet') this.magnetPull(dt)
   410→   410→
   411→   411→    // Радиус подбора зависит от перка лапки
   412→   412→    const grabR = (this.perk === 'bigpaw' ? 58 : this.perk === 'smallpaw' ? 34 : 44) * u
   413→   413→
   414→   414→    // Подбор попкорна / ловушка-кукуруза
   415→   415→    for (let i = this.pieces.length - 1; i >= 0; i--) {
   416→   416→      const p = this.pieces[i]
   417→   417→      if (p.spawn > 0.12) continue
   418→   418→      const d = Math.hypot(this.paw.x - p.x, this.paw.y - p.y)
   419→   419→      if (p.kind === 'corn') {
   420→   420→        // Кукуруза — мина: задел = минус балл
   421→   421→        if (d < 38 * u) {
   422→   422→          this.pieces.splice(i, 1)
   423→   423→          this.score = Math.max(0, this.score - 1)
   424→   424→          this.paw.hurt = 1
   425→   425→          this.texts.push({
   426→   426→            x: this.paw.x,
   427→   427→            y: this.paw.y - 34 * u,
   428→   428→            text: '-1',
   429→   429→            color: '#FF6B5E',
   430→   430→            t: 0,
   431→   431→            dur: 0.9,
   432→   432→            size: 24 * u,
   433→   433→          })
   434→   434→          this.crumbs(p.x, p.y, '#E8960C', 7)
   435→   435→          audio.play('penalty')
   436→   436→        }
   437→   437→        continue
   438→   438→      }
   439→   439→      if (d < grabR && this.carried.length < CARRY_MAX) {
   440→   440→        this.pieces.splice(i, 1)
   441→   441→        this.carried.push({ golden: p.kind === 'golden', grabT: 0 })
   442→   442→        this.paw.pulse = 1
   443→   443→        this.crumbs(p.x, p.y, p.kind === 'golden' ? '#FFE066' : '#FFF3D6', 6)
   444→   444→        audio.play(p.kind === 'golden' ? 'golden' : 'pickup')
   445→   445→      }
   446→   446→    }
   447→   447→
   448→   448→    // Высыпание в стаканчик
   449→   449→    if (this.carried.length > 0) {
   450→   450→      const dc = Math.hypot(this.paw.x - this.cup.x, this.paw.y - this.cup.y)
   451→   451→      if (dc < 78 * u) {
   452→   452→        let base = 0
   453→   453→        let hasGold = false
   454→   454→        for (const c of this.carried) {
   455→   455→          base += c.golden ? 5 : 1
   456→   456→          hasGold = hasGold || c.golden
   457→   457→        }
   458→   458→        const mult = this.perk === 'doublepoints' ? 2 : 1
   459→   459→        const sum = base * mult
   460→   460→        this.collected += this.carried.length
   461→   461→        this.carried.forEach((c, i) => {
   462→   462→          this.flying.push({
   463→   463→            x0: this.paw.x + (i - 1) * 8 * u,
   464→   464→            y0: this.paw.y - 20 * u - i * 10 * u,
   465→   465→            golden: c.golden,
   466→   466→            t: 0,
   467→   467→          })
   468→   468→        })
   469→   469→        this.carried = []
   470→   470→        this.score += sum
   471→   471→        this.cup.squash = 1
   472→   472→        this.texts.push({
   473→   473→          x: this.cup.x,
   474→   474→          y: this.cup.y - this.cupH * 0.8,
   475→   475→          text: `+${sum}`,
   476→   476→          color: hasGold ? '#FFC93C' : '#FFFFFF',
   477→   477→          t: 0,
   478→   478→          dur: 0.9,
   479→   479→          size: 26 * u,
   480→   480→        })
   481→   481→        audio.play('deposit')
   482→   482→      }
   483→   483→    }
   484→   484→
   485→   485→    // Летящие в стаканчик зёрна
   486→   486→    for (let i = this.flying.length - 1; i >= 0; i--) {
   487→   487→      const f = this.flying[i]
   488→   488→      f.t += dt / 0.3
   489→   489→      if (f.t >= 1) {
   490→   490→        this.flying.splice(i, 1)
   491→   491→        this.crumbs(this.cup.x, this.cup.y - this.cupH * 0.5, '#FFF6EC', 3)
   492→   492→      }
   493→   493→    }
   494→   494→
   495→   495→    // Вся волна собрана → следующая. Кукуруза-мины НЕ мешает: считаем только попкорн
   496→   496→    const hasPopcorn = this.pieces.some((p) => p.kind !== 'corn')
   497→   497→    if (
   498→   498→      this.waveDelay < 0 &&
   499→   499→      !hasPopcorn &&
   500→   500→      this.carried.length === 0 &&
   501→   501→      this.flying.length === 0
   502→   502→    ) {
   503→   503→      this.waveDelay = this.baseWaveDelay
   504→   504→    }
   505→   505→    if (this.waveDelay >= 0) {
   506→   506→      this.waveDelay -= dt
   507→   507→      if (this.waveDelay <= 0) {
   508→   508→        this.waveDelay = -1
   509→   509→        this.wave++
   510→   510→        this.banner = { text: `ВОЛНА ${this.wave}`, t: 0 }
   511→   511→        audio.play('wave')
   512→   512→        this.spawnWave()
   513→   513→      }
   514→   514→    }
   515→   515→
   516→   516→    // Норма выполнена → короткая пауза (зёрна долетают) → уровень пройден
   517→   517→    if (this.pendingClear < 0 && this.collected >= this.quota) {
   518→   518→      this.pendingClear = CLEAR_DELAY
   519→   519→    }
   520→   520→    if (this.pendingClear >= 0) {
   521→   521→      this.pendingClear -= dt
   522→   522→      if (this.pendingClear <= 0) {
   523→   523→        this.levelComplete()
   524→   524→        return
   525→   525→      }
   526→   526→    }
   527→   527→
   528→   528→    // Музыка зависит от оставшегося времени: >30с база, ≤30с +25%, ≤15с +40%, ≤5с 240 BPM
   529→   529→    const stage = this.timeLeft > 30 ? 0 : this.timeLeft > 15 ? 1 : this.timeLeft > 5 ? 2 : 3
   530→   530→    if (stage !== this.audioStage) {
   531→   531→      this.audioStage = stage
   532→   532→      audio.setIntensity(stage)
   533→   533→    }
   534→   534→
   535→   535→    // Таймер
   536→   536→    this.timeLeft -= dt
   537→   537→    const sec = Math.max(0, Math.ceil(this.timeLeft))
   538→   538→    if (sec !== this.lastSec) {
   539→   539→      this.lastSec = sec
   540→   540→      if (sec <= 5 && sec > 0) audio.play('tick')
   541→   541→    }
   542→   542→    if (this.timeLeft <= 0) {
   543→   543→      this.timeLeft = 0
   544→   544→      this.end()
   545→   545→      return
   546→   546→    }
   547→   547→
   548→   548→    // Косметика
   549→   549→    this.paw.pulse = Math.max(0, this.paw.pulse - dt * 5)
   550→   550→    this.paw.hurt = Math.max(0, this.paw.hurt - dt * 2.2)
   551→   551→    this.cup.squash = Math.max(0, this.cup.squash - dt * 4)
   552→   552→    this.banner.t += dt
   553→   553→    for (const p of this.pieces) p.spawn = Math.max(0, p.spawn - dt)
   554→   554→    for (const c of this.carried) c.grabT = Math.min(1, c.grabT + dt * 8)
   555→   555→    for (let i = this.particles.length - 1; i >= 0; i--) {
   556→   556→      const p = this.particles[i]
   557→   557→      p.t += dt
   558→   558→      p.vy += 900 * u * dt
   559→   559→      p.x += p.vx * dt
   560→   560→      p.y += p.vy * dt
   561→   561→      if (p.t >= p.ttl) this.particles.splice(i, 1)
   562→   562→    }
   563→   563→    for (let i = this.texts.length - 1; i >= 0; i--) {
   564→   564→      const t = this.texts[i]
   565→   565→      t.t += dt
   566→   566→      if (t.t >= t.dur) this.texts.splice(i, 1)
   567→   567→    }
   568→   568→  }
   569→   569→
   570→   570→  private levelComplete(): void {
   571→   571→    this.phase = 'clear'
   572→   572→    const bonus = Math.ceil(this.timeLeft)
   573→   573→    this.score += bonus
   574→   574→    audio.stopMusic()
   575→   575→    audio.play('levelwin')
   576→   576→    this.emitStats(true)
   577→   577→    this.cb.onLevelComplete?.({ level: this.lvl, score: this.score, bonus })
   578→   578→  }
   579→   579→
   580→   580→  private end(): void {
   581→   581→    this.phase = 'ended'
   582→   582→    audio.stopMusic()
   583→   583→    audio.play('gameover')
   584→   584→    this.emitStats(true)
   585→   585→    this.cb.onEnd?.({ score: this.score, level: this.lvl })
   586→   586→  }
   587→   587→
   588→   588→  private crumbs(x: number, y: number, color: string, n: number): void {
   589→   589→    const u = this.u
   590→   590→    for (let i = 0; i < n; i++) {
   591→   591→      const a = Math.random() * Math.PI * 2
   592→   592→      const sp = (60 + Math.random() * 120) * u
   593→   593→      this.particles.push({
   594→   594→        x,
   595→   595→        y,
   596→   596→        vx: Math.cos(a) * sp,
   597→   597→        vy: Math.sin(a) * sp - 60 * u,
   598→   598→        t: 0,
   599→   599→        ttl: 0.35 + Math.random() * 0.25,
   600→   600→        color,
   601→   601→        size: (2 + Math.random() * 2.5) * u,
   602→   602→      })
   603→   603→    }
   604→   604→  }
   605→   605→
   606→   606→  /* ================= цикл ================= */
   607→   607→
   608→   608→  private loop = (ts: number): void => {
   609→   609→    if (this.destroyed) return
   610→   610→    const dt = Math.min(0.05, (ts - this.lastTs) / 1000)
   611→   611→    this.lastTs = ts
   612→   612→    if (this.phase === 'playing' && !this.paused) {
   613→   613→      try {
   614→   614→        this.update(dt)
   615→   615→      } catch (err) {
   616→   616→        // Сбой кадра не должен замораживать игру — логируем и продолжаем
   617→   617→        console.error('[popcorn] update error:', err)
   618→   618→      }
   619→   619→    }
   620→   620→    this.render(ts)
   621→   621→    this.emitStats(false)
   622→   622→    this.raf = requestAnimationFrame(this.loop)
   623→   623→  }
   624→   624→
   625→   625→  private emitStats(force: boolean): void {
   626→   626→    if (!this.cb.onStats) return
   627→   627→    const sec = Math.ceil(this.timeLeft)
   628→   628→    const key = `${this.score}|${this.lvl}|${this.quota}|${this.collected}|${this.carried.length}|${sec}|${this.paused}`
   629→   629→    if (!force && key === this.lastStatsKey) return
   630→   630→    this.lastStatsKey = key
   631→   631→    this.cb.onStats({
   632→   632→      score: this.score,
   633→   633→      timeLeft: this.timeLeft,
   634→   634→      duration: this.duration,
   635→   635→      level: this.lvl,
   636→   636→      quota: this.quota,
   637→   637→      collected: this.collected,
   638→   638→      carrying: this.carried.length,
   639→   639→      paused: this.paused,
   640→   640→    })
   641→   641→  }

   645→   645→  private render(ts: number): void {
   646→   646→    const { ctx } = this
   647→   647→    ctx.clearRect(0, 0, this.W, this.H)
   648→   648→    if (this.phase === 'idle') return
   649→   649→
   650→   650→    this.drawCup()
   651→   651→    this.drawPieces(ts)
   652→   652→    this.drawFlying()
   653→   653→    this.drawParticles()
   654→   654→    this.drawPaw(ts)
   655→   655→    this.drawTexts()
   656→   656→    this.drawBanner()
   657→   657→    if (this.perk === 'fog' && this.phase === 'playing') this.drawFog()
   658→   658→
   659→   659→    if (this.phase === 'ended' || this.phase === 'clear') {
   660→   660→      ctx.fillStyle = 'rgba(24, 10, 4, 0.35)'
   661→   661→      ctx.fillRect(0, 0, this.W, this.H)
   662→   662→    }
   663→   663→  }

   665→   665→  /** «Попкорнный туман»: ступенчатые кольца темноты вокруг лапки (8-битный стиль) */
   666→   666→  private drawFog(): void {
   667→   667→    const { ctx } = this
   668→   668→    const u = this.u
   669→   669→    const rings: Array<[number, number]> = [
   670→   670→      [150, 0.86],
   671→   671→      [120, 0.8],
   672→   672→      [95, 0.66],
   673→   673→      [72, 0.45],
   674→   674→      [52, 0.22],
   675→   675→    ]
   676→   676→    for (const [r, a] of rings) {
   677→   677→      ctx.fillStyle = `rgba(12, 6, 2, ${a})`
   678→   678→      ctx.beginPath()
   679→   679→      ctx.rect(0, 0, this.W, this.H)
   680→   680→      ctx.arc(this.paw.x, this.paw.y, r * u, 0, Math.PI * 2)
   681→   681→      ctx.fill('evenodd')
   682→   682→    }
   683→   683→  }

   685→   685→  private drawCup(): void {
   686→   686→    const { ctx } = this
   687→   687→    const s = this.cupScale
   688→   688→    const sq = this.cup.squash
   689→   689→    const sx = 1 + 0.08 * sq
   690→   690→    const sy = 1 - 0.1 * sq
   691→   691→
   692→   692→    // Тень
   693→   693→    ctx.fillStyle = 'rgba(0, 0, 0, 0.22)'
   694→   694→    ctx.beginPath()
   695→   695→    ctx.ellipse(
   696→   696→      this.cup.x,
   697→   697→      this.cup.y + (this.cupH / 2) * sy + 5 * this.u,
   698→   698→      48 * this.u * sx,
   699→   699→      10 * this.u,
   700→   700→      0,
   701→   701→      0,
   702→   702→      Math.PI * 2,
   703→   703→    )
   704→   704→    ctx.fill()
   705→   705→
   706→   706→    // «Шапка» попкорна растёт вместе с выполнением нормы
   707→   707→    const progress = this.quota > 0 ? this.collected / this.quota : 0
   708→   708→    const puffs = Math.min(PUFF_POS.length, Math.ceil(progress * PUFF_POS.length))
   709→   709→    const mouthY = this.cup.y - 13 * s * sy
   710→   710→    for (let i = 0; i < puffs; i++) {
   711→   711→      const [ox, oy] = PUFF_POS[i]
   712→   712→      drawSprite(ctx, this.sp.popcorn, this.cup.x + ox * s * 0.85, mouthY + oy * s * 0.75, s * 0.52, {
   713→   713→        rot: i * 0.7,
   714→   714→      })
   715→   715→    }
   716→   716→
   717→   717→    // Корпус стаканчика
   718→   718→    drawSprite(ctx, this.sp.cup, this.cup.x, this.cup.y, s, { sx, sy })
   719→   719→  }

   721→   721→  private drawPieces(ts: number): void {
   722→   722→    const { ctx } = this
   723→   723→    const u = this.u
   724→   724→    for (const p of this.pieces) {
   725→   725→      // Мины-зерна статичны (никаких прыжков) — попкорн слегка «дышит»
   726→   726→      const bobAmp = p.kind === 'corn' ? 0 : 2.5
   727→   727→      const bob = Math.sin(ts * 0.004 + p.phase) * bobAmp * u
   728→   728→      const y = p.y + bob
   729→   729→      const pop = p.spawn > 0 ? easeOutBack(1 - p.spawn / (SPAWN_DUR + 0.5)) : 1
   730→   730→      const base = p.kind === 'corn' ? 2.3 : 2.6
   731→   731→      const scale = base * u * Math.max(0.05, pop)
   732→   732→
   733→   733→      // Тень
   734→   734→      ctx.fillStyle = 'rgba(0, 0, 0, 0.16)'
   735→   735→      ctx.beginPath()
   736→   736→      ctx.ellipse(p.x, p.y + 9 * u, 12 * u * pop, 4 * u * pop, 0, 0, Math.PI * 2)
   737→   737→      ctx.fill()
   738→   738→
   739→   739→      const sprite = p.kind === 'corn' ? this.sp.corn : p.kind === 'golden' ? this.sp.golden : this.sp.popcorn
   740→   740→      drawSprite(ctx, sprite, p.x, y, scale, { rot: p.rot })
   741→   741→      if (p.kind === 'golden') {
   742→   742→        drawSprite(ctx, this.sp.sparkle, p.x + 10 * u, y - 11 * u, 1.15 * u, {
   743→   743→          rot: ts * 0.004,
   744→   744→          alpha: 0.95,
   745→   745→        })
   746→   746→      }
   747→   747→    }
   748→   748→  }
   749→   749→
   750→   750→  private drawFlying(): void {
   751→   751→    const { ctx } = this
   752→   752→    const u = this.u
   753→   753→    const tx = this.cup.x
   754→   754→    const ty = this.cup.y - 10 * this.cupScale
   755→   755→    for (const f of this.flying) {
   756→   756→      const q = easeOutQuad(f.t)
   757→   757→      const x = f.x0 + (tx - f.x0) * q
   758→   758→      const y = f.y0 + (ty - f.y0) * q - Math.sin(q * Math.PI) * 30 * u
   759→   759→      drawSprite(ctx, f.golden ? this.sp.golden : this.sp.popcorn, x, y, 2.6 * u * (1 - 0.45 * q), {
   760→   760→        rot: q * 3,
   761→   761→      })
   762→   762→    }
   763→   763→  }
   764→   764→
   765→   765→  private drawPaw(ts: number): void {
   766→   766→    const { ctx } = this
   767→   767→    const u = this.u
   768→   768→    if (!this.paw.seen && this.phase === 'idle') return
   769→   769→
   770→   770→    // Тряска при штрафе + сильная дрожь перка «Нервная лапа»
   771→   771→    const shake = Math.sin(ts * 0.055) * 5 * this.paw.hurt
   772→   772→    const shiver = this.perk === 'shaky'
   773→   773→    const jx = shiver ? Math.sin(ts * 0.021) * 9 * u + Math.sin(ts * 0.047) * 6 * u : 0
   774→   774→    const jy = shiver ? Math.cos(ts * 0.017) * 7 * u + Math.sin(ts * 0.053) * 4 * u : 0
   775→   775→    const px = this.paw.x + shake + jx
   776→   776→    const py = this.paw.y + jy
   777→   777→
   778→   778→    // Тень лапки
   779→   779→    ctx.fillStyle = 'rgba(0, 0, 0, 0.18)'
   780→   780→    ctx.beginPath()
   781→   781→    ctx.ellipse(px, py + 28 * u, 26 * u, 7 * u, 0, 0, Math.PI * 2)
   782→   782→    ctx.fill()
   783→   783→
   784→   784→    const basePaw = this.perk === 'bigpaw' ? 4.8 : this.perk === 'smallpaw' ? 2.9 : 3.8
   785→   785→    const s = basePaw * u * (1 + 0.13 * this.paw.pulse)
   786→   786→    drawSprite(ctx, this.sp.paw, px, py, s)
   787→   787→
   788→   788→    // Зёрна в лапке (стопкой над подушечкой)
   789→   789→    this.carried.forEach((c, i) => {
   790→   790→      const wig = Math.sin(ts * 0.012 + i * 2) * 1.6 * u
   791→   791→      const y = py - 32 * u - i * 15 * u + (1 - c.grabT) * 10 * u
   792→   792→      drawSprite(ctx, c.golden ? this.sp.golden : this.sp.popcorn, px + wig, y, 2.3 * u, {
   793→   793→        rot: Math.sin(ts * 0.006 + i) * 0.12,
   794→   794→      })
   795→   795→    })
   796→   796→  }
   797→   797→
   798→   798→  private drawParticles(): void {
   799→   799→    const { ctx } = this
   800→   800→    for (const p of this.particles) {
   801→   801→      ctx.globalAlpha = 1 - p.t / p.ttl
   802→   802→      ctx.fillStyle = p.color
   803→   803→      ctx.fillRect(p.x - p.size / 2, p.y - p.size / 2, p.size, p.size)
   804→   804→    }
   805→   805→    ctx.globalAlpha = 1
   806→   806→  }
   807→   807→
   808→   808→  private drawTexts(): void {
   809→   809→    const { ctx } = this
   810→   810→    for (const t of this.texts) {
   811→   811→      const q = t.t / t.dur
   812→   812→      const alpha = q < 0.15 ? q / 0.15 : 1 - Math.max(0, (q - 0.55) / 0.45)
   813→   813→      const y = t.y - 42 * this.u * q
   814→   814→      ctx.save()
   815→   815→      ctx.globalAlpha = Math.max(0, Math.min(1, alpha))
   816→   816→      ctx.font = `600 ${t.size}px "Handjet", system-ui, sans-serif`
   817→   817→      ctx.textAlign = 'center'
   818→   818→      ctx.textBaseline = 'middle'
   819→   819→      ctx.lineWidth = Math.max(3, t.size * 0.16)
   820→   820→      ctx.strokeStyle = 'rgba(46, 18, 6, 0.9)'
   821→   821→      ctx.strokeText(t.text, t.x, y)
   822→   822→      ctx.fillStyle = t.color
   823→   823→      ctx.fillText(t.text, t.x, y)
   824→   824→      ctx.restore()
   825→   825→    }
   826→   826→  }
   827→   827→
   828→   828→  private drawBanner(): void {
   829→   829→    if (this.banner.t > 1.5) return
   830→   830→    const { ctx } = this
   831→   831→    const t = this.banner.t
   832→   832→    const alpha = t < 0.2 ? t / 0.2 : t > 1.1 ? Math.max(0, (1.5 - t) / 0.4) : 1
   833→   833→    const pop = 1 + Math.max(0, 1 - t / 0.25) * 0.5
   834→   834→    const size = 34 * this.u * pop
   835→   835→    ctx.save()
   836→   836→    ctx.globalAlpha = alpha
   837→   837→    ctx.font = `700 ${size}px "Handjet", system-ui, sans-serif`
   838→   838→    ctx.textAlign = 'center'
   839→   839→    ctx.textBaseline = 'middle'
   840→   840→    ctx.lineWidth = Math.max(4, size * 0.18)
   841→   841→    ctx.strokeStyle = 'rgba(46, 18, 6, 0.9)'
   842→   842→    ctx.strokeText(this.banner.text, this.W / 2, this.H * 0.3)
   843→   843→    ctx.fillStyle = '#FFE066'
   844→   844→    ctx.fillText(this.banner.text, this.W / 2, this.H * 0.3)
   845→   845→    ctx.restore()
   846→   846→  }
