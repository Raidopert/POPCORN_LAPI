    53→    53→/* ---------- localStorage как внешний стор (без каскадных рендеров) ---------- */
    54→    54→
    55→    55→function subscribeStore(onChange: () => void): () => void {
    56→    56→  window.addEventListener('storage', onChange)
    57→    57→  window.addEventListener(STORE_EVENT, onChange)
    58→    58→  return () => {
    59→    59→    window.removeEventListener('storage', onChange)
    60→    60→    window.removeEventListener(STORE_EVENT, onChange)
    61→    61→  }
    62→    62→}
    63→    63→
    64→    64→function useStoredValue(key: string, fallback: string): [string, (v: string) => void] {
    65→    65→  const value = useSyncExternalStore(
    66→    66→    subscribeStore,
    67→    67→    () => window.localStorage.getItem(key) ?? fallback,
    68→    68→    () => fallback,
    69→    69→  )
    70→    70→  const setValue = useCallback(
    71→    71→    (v: string) => {
    72→    72→      window.localStorage.setItem(key, v)
    73→    73→      window.dispatchEvent(new Event(STORE_EVENT))
    74→    74→    },
    75→    75→    [key],
    76→    76→  )
    77→    77→  return [value, setValue]
    78→    78→}

   177→   177→function ChestImg({ open }: { open: boolean }) {
   178→   178→  const ref = useRef<HTMLCanvasElement | null>(null)
   179→   179→  useEffect(() => {
   180→   180→    const c = ref.current
   181→   181→    if (!c) return
   182→   182→    const spr = makeChestSprite(open)
   183→   183→    const px = 6
   184→   184→    c.width = spr.w * px
   185→   185→    c.height = spr.h * px
   186→   186→    const ctx = c.getContext('2d')
   187→   187→    if (!ctx) return
   188→   188→    ctx.imageSmoothingEnabled = false
   189→   189→    ctx.clearRect(0, 0, c.width, c.height)
   190→   190→    ctx.drawImage(spr.canvas, 0, 0, c.width, c.height)
   191→   191→  }, [open])
   192→   192→  return <canvas ref={ref} aria-hidden className="pixelated h-auto w-16 sm:w-20" />
   193→   193→}
