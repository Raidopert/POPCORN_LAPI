    39→/* ---------- localStorage как внешний стор (без каскадных рендеров) ---------- */
    40→
    41→function subscribeStore(onChange: () => void): () => void {
    42→  window.addEventListener('storage', onChange)
    43→  window.addEventListener(STORE_EVENT, onChange)
    44→  return () => {
    45→    window.removeEventListener('storage', onChange)
    46→    window.removeEventListener(STORE_EVENT, onChange)
    47→  }
    48→}
    49→
    50→function useStoredValue(key: string, fallback: string): [string, (v: string) => void] {
    51→  const value = useSyncExternalStore(
    52→    subscribeStore,
    53→    () => window.localStorage.getItem(key) ?? fallback,
    54→    () => fallback,
    55→  )
    56→  const setValue = useCallback(
    57→    (v: string) => {
    58→      window.localStorage.setItem(key, v)
    59→      window.dispatchEvent(new Event(STORE_EVENT))
    60→    },
    61→    [key],
    62→  )
    63→  return [value, setValue]
    64→}

   162→/** Пиксельный сундучок (закрытый/открытый), отрисованный на canvas */
   163→function ChestImg({ open }: { open: boolean }) {
   164→  const ref = useRef<HTMLCanvasElement | null>(null)
   165→  useEffect(() => {
   166→    const c = ref.current
   167→    if (!c) return
   168→    const spr = makeChestSprite(open)
   169→    const px = 6
   170→    c.width = spr.w * px
   171→    c.height = spr.h * px
   172→    const ctx = c.getContext('2d')
   173→    if (!ctx) return
   174→    ctx.imageSmoothingEnabled = false
   175→    ctx.clearRect(0, 0, c.width, c.height)
   176→    ctx.drawImage(spr.canvas, 0, 0, c.width, c.height)
   177→  }, [open])
   178→  return <canvas ref={ref} aria-hidden className="pixelated h-auto w-16 sm:w-20" />
   179→}

   183→export function PopcornGame() {
   184→  const mainRef = useRef<HTMLElement | null>(null)
   185→  const canvasRef = useRef<HTMLCanvasElement | null>(null)
   186→  const engineRef = useRef<PopcornEngine | null>(null)
   187→  const mutedRef = useRef(false)
   188→  const transformRef = useRef<ScreenTransform>('none')
   189→  const layoutRef = useRef<() => void>(() => {})
   190→  const devRef = useRef<{ fast: boolean; forced: PerkId | null }>({ fast: false, forced: null })
   191→
   192→  const [phase, setPhase] = useState<Phase>('start')
   193→  const [stats, setStats] = useState<EngineStats>({
   194→    score: 0,
   195→    timeLeft: 60,
   196→    duration: 60,
   197→    level: 1,
   198→    quota: 9,
   199→    collected: 0,
   200→    carrying: 0,
   201→    carryMax: 3,
   202→    freezeLeft: 0,
   203→    paused: false,
   204→  })
   205→  const [bestRaw, storeBest] = useStoredValue(BEST_KEY, '0')
   206→  const [mutedRaw, storeMuted] = useStoredValue(MUTE_KEY, '0')
   207→  const best = Number(bestRaw) || 0
   208→  const muted = mutedRaw === '1'
   209→  const [result, setResult] = useState<ResultState | null>(null)
   210→  const [clearInfo, setClearInfo] = useState<ClearState | null>(null)
   211→  const [offers, setOffers] = useState<PerkDef[]>([])
   212→  const [openedIdx, setOpenedIdx] = useState<number | null>(null)
   213→  const [runLevel, setRunLevel] = useState(2)
   214→  const [perk, setPerk] = useState<PerkId | null>(null)
   215→
   216→  /* Скрытые dev-параметры URL: ?fast=1 (короткая норма) и ?perk=rot180 */
   217→  useEffect(() => {
   218→    const sp = new URLSearchParams(window.location.search)
   219→    devRef.current = {
   220→      fast: sp.has('fast'),
   221→      forced: (sp.get('perk') as PerkId | null) ?? null,
   222→    }
   223→  }, [])
   224→
   225→  /* Синхронизируем состояние звука с аудио-движком (внешняя система) */
   226→  useEffect(() => {
   227→    mutedRef.current = muted
   228→    audio.setMuted(muted)
   229→  }, [muted])
   230→
   231→  /* Раскладка холста с учётом трансформации перка (ротация/зеркало) */
   232→  const applyLayout = useCallback(() => {
   233→    const main = mainRef.current
   234→    const canvas = canvasRef.current
   235→    if (!main || !canvas) return
   236→    const cw = main.clientWidth
   237→    const ch = main.clientHeight
   238→    if (cw === 0 || ch === 0) return
   239→    const t = transformRef.current
   240→    const rot90 = t === 'rot90'
   241→    const gw = rot90 ? ch : cw
   242→    const gh = rot90 ? cw : ch
   243→    engineRef.current?.handleResize(gw, gh)
   244→    canvas.style.width = `${gw}px`
   245→    canvas.style.height = `${gh}px`
   246→    if (rot90) {
   247→      canvas.style.left = '50%'
   248→      canvas.style.top = '50%'
   249→      canvas.style.transform = 'translate(-50%, -50%) rotate(90deg)'
   250→    } else {
   251→      canvas.style.left = '0'
   252→      canvas.style.top = '0'
   253→      canvas.style.transform =
   254→        t === 'rot180' ? 'rotate(180deg)' : t === 'mirror' ? 'scaleX(-1)' : 'none'
   255→    }
   256→  }, [])
   257→
   258→  useEffect(() => {
   259→    layoutRef.current = applyLayout
   260→  }, [applyLayout])
   261→
   262→  /* Инициализация движка */
   263→  useEffect(() => {
   264→    const canvas = canvasRef.current
   265→    if (!canvas) return
   266→
   267→    const engine = new PopcornEngine(canvas, {
   268→      onStats: (s) => setStats(s),
   269→      onLevelComplete: (r) => {
   270→        setClearInfo({ level: r.level, bonus: r.bonus })
   271→        setOffers(rollPerkOffer(3, devRef.current.forced ?? undefined))
   272→        setOpenedIdx(null)
   273→        setRunLevel(r.level + 1)
   274→        setPhase('chest')
   275→      },
   276→      onEnd: (r) => {
   277→        const prevBest = Number(window.localStorage.getItem(BEST_KEY) ?? 0)
   278→        const record = r.score > prevBest
   279→        if (record) {
   280→          storeBest(String(r.score))
   281→          window.setTimeout(() => audio.play('record'), 700)
   282→        }
   283→        setResult({ score: r.score, level: r.level, record })
   284→        setPhase('ended')
   285→      },
   286→    })
   287→    engineRef.current = engine
   288→    ;(window as unknown as { __popcornEngine?: PopcornEngine }).__popcornEngine = engine
   289→    layoutRef.current()
   290→
   291→    return () => {
   292→      engine.destroy()
   293→      engineRef.current = null
   294→      delete (window as unknown as { __popcornEngine?: PopcornEngine }).__popcornEngine
   295→    }
   296→  }, [storeBest])
   297→
   298→  /* Следим за размером контейнера */
   299→  useEffect(() => {
   300→    const main = mainRef.current
   301→    if (!main || typeof ResizeObserver === 'undefined') return
   302→    const ro = new ResizeObserver(() => layoutRef.current())
   303→    ro.observe(main)
   304→    return () => ro.disconnect()
   305→  }, [])
   306→
   307→  /* Ввод: инверсия управления для перков-трансформаций.
   308→     rot180/mirror — сырые координаты: CSS-зеркало переворачивает картинку,
   309→     и лапка на экране идёт в противоположную сторону от пальца.
   310→     rot90 — мир повёрнут с заменой измерений, поэтому даём развёрнутую инверсию. */
   311→  useEffect(() => {
   312→    const main = mainRef.current
   313→    if (!main) return
   314→    const onPointer = (e: PointerEvent) => {
   315→      const eng = engineRef.current
   316→      if (!eng) return
   317→      const r = main.getBoundingClientRect()
   318→      const sx = e.clientX - r.left
   319→      const sy = e.clientY - r.top
   320→      let x = sx
   321→      let y = sy
   322→      if (transformRef.current === 'rot90') {
   323→        x = r.height - sy
   324→        y = sx
   325→      }
   326→      eng.setPointer(x, y, e.pointerType === 'touch' ? 64 : 0)
   327→    }
   328→    main.addEventListener('pointerdown', onPointer)
   329→    main.addEventListener('pointermove', onPointer)
   330→    return () => {
   331→      main.removeEventListener('pointerdown', onPointer)
   332→      main.removeEventListener('pointermove', onPointer)
   333→    }
   334→  }, [])
   335→
   336→  /* Трансформации экрана и фильтры действуют только во время уровня */
   337→  useEffect(() => {
   338→    const onLevel = phase === 'playing' || phase === 'paused'
   339→    transformRef.current = onLevel ? transformFor(perk) : 'none'
   340→    const canvas = canvasRef.current
   341→    if (canvas) {
   342→      canvas.style.filter = onLevel && perk === 'grayscale' ? 'grayscale(1) contrast(1.05)' : ''
   343→    }
   344→    layoutRef.current()
   345→  }, [perk, phase])
   346→
   347→  /* Автопауза при сворачивании вкладки: игру и музыку ставим на паузу,
   348→     чтобы троттлинг таймеров не копил отставание аудио-планировщика */
   349→  useEffect(() => {
   350→    const onVisibility = () => {
   351→      if (document.hidden) {
   352→        audio.stopMusic()
   353→        engineRef.current?.setPaused(true)
   354→        setPhase((p) => (p === 'playing' ? 'paused' : p))
   355→      } else if (engineRef.current?.phase === 'playing') {
   356→        audio.startMusic()
   357→      }
   358→    }
   359→    document.addEventListener('visibilitychange', onVisibility)
   360→    return () => document.removeEventListener('visibilitychange', onVisibility)
   361→  }, [])
   362→
   363→  const startGame = useCallback(() => {
   364→    setResult(null)
   365→    setClearInfo(null)
   366→    setPerk(null)
   367→    transformRef.current = 'none'
   368→    const canvas = canvasRef.current
   369→    if (canvas) canvas.style.filter = ''
   370→    audio.ensure()
   371→    audio.setMuted(mutedRef.current)
   372→    audio.startMusic()
   373→    audio.play('click')
   374→    engineRef.current?.startRun({ fast: devRef.current.fast })
   375→    setPhase('playing')
   376→  }, [])
   377→
   378→  const beginLevel = useCallback((lvl: number, perkId: PerkId | null) => {
   379→    audio.ensure()
   380→    audio.setMuted(mutedRef.current)
   381→    audio.startMusic()
   382→    audio.play('click')
   383→    transformRef.current = transformFor(perkId)
   384→    const canvas = canvasRef.current
   385→    if (canvas) {
   386→      canvas.style.filter = perkId === 'grayscale' ? 'grayscale(1) contrast(1.05)' : ''
   387→    }
   388→    layoutRef.current()
   389→    engineRef.current?.startLevel(lvl, perkId)
   390→    setPerk(perkId)
   391→    setPhase('playing')
   392→  }, [])
   393→
   394→  const resumeGame = useCallback(() => {
   395→    audio.play('click')
   396→    audio.startMusic()
   397→    engineRef.current?.setPaused(false)
   398→    setPhase('playing')
   399→  }, [])
   400→
   401→  const pauseGame = useCallback(() => {
   402→    audio.play('click')
   403→    audio.stopMusic()
   404→    engineRef.current?.setPaused(true)
   405→    setPhase('paused')
   406→  }, [])
   407→
   408→  const toggleMute = useCallback(() => {
   409→    storeMuted(muted ? '0' : '1')
   410→  }, [muted, storeMuted])
   411→
   412→  const openChest = useCallback((i: number) => {
   413→    audio.play('chest')
   414→    setOpenedIdx(i)
   415→    window.setTimeout(() => audio.play('perk'), 380)
   416→  }, [])
   417→
   418→  const sec = Math.ceil(stats.timeLeft)
   419→  const urgent = sec <= 10
   420→  const playing = phase === 'playing' || phase === 'paused'
   421→  const quotaPct = Math.min(100, (stats.collected / Math.max(1, stats.quota)) * 100)
   422→  const chosen: PerkDef | null = openedIdx !== null ? (offers[openedIdx] ?? null) : null
   423→  const activePerk = perk ? PERK_MAP[perk] : null
   424→
   425→  return (
   426→    <div className="flex h-dvh flex-col overflow-hidden bg-[#1d110b] select-none">
   427→      <main
   428→        ref={mainRef}
   429→        className="relative min-h-0 flex-1 touch-none overflow-hidden"
   430→        aria-label="Игровое поле: соберите рассыпанный попкорн в стаканчик лапкой куницы"
   431→      >
   432→        {/* Фон — ретро комната */}
   433→        <div
   434→          aria-hidden
   435→          className="absolute inset-0 scale-105 bg-[url('/popcorn-room.png')] bg-cover bg-center"
   436→        />
   437→        <div
   438→          aria-hidden
   439→          className="absolute inset-0 bg-gradient-to-b from-black/50 via-black/10 to-black/55"
   440→        />
   441→
   442→        {/* Игровой холст (трансформации перков применяются инлайн) */}
   443→        <canvas
   444→          ref={canvasRef}
   445→          className="absolute z-10 touch-none"
   446→          onContextMenu={(e) => e.preventDefault()}
   447→        />
   448→
   449→        {/* Фриз-тайм кота: большие красные цифры отсчёта на фоне */}
   450→        {playing && stats.freezeLeft > 0 && (
   451→          <div
   452→            aria-hidden
   453→            className="pointer-events-none absolute inset-0 z-[15] flex flex-col items-center justify-center gap-2"
   454→          >
   455→            <span className="rounded-full bg-black/45 px-3 py-1 font-pixel text-sm font-bold tracking-widest text-emerald-100 ring-1 ring-emerald-200/40 sm:text-base">
   456→              ТАЙМЕР ЗАМОРОЖЕН · ПРОДУМАЙ МАРШРУТ
   457→            </span>
   458→            <AnimatePresence mode="popLayout">
   459→              <motion.span
   460→                key={Math.ceil(stats.freezeLeft)}
   461→                initial={{ scale: 1.7, opacity: 0 }}
   462→                animate={{ scale: 1, opacity: 1 }}
   463→                exit={{ scale: 0.6, opacity: 0 }}
   464→                transition={{ duration: 0.16 }}
   465→                className="font-pixel text-[44vw] font-bold leading-none text-red-500/90 drop-shadow-[0_6px_0_rgba(0,0,0,0.45)] sm:text-[230px]"
   466→              >
   467→                {Math.ceil(stats.freezeLeft)}
   468→              </motion.span>
   469→            </AnimatePresence>
   470→          </div>
   471→        )}
   472→
   473→        {/* HUD */}
   474→        <div className="pointer-events-none absolute inset-x-0 top-0 z-20 flex items-start justify-between gap-2 p-3 pt-[max(0.75rem,env(safe-area-inset-top))]">
   475→          <div className="flex flex-col gap-2">
   476→            {playing ? (
   477→              <>
   478→                <Pill>
   479→                  <Popcorn className="h-4 w-4 text-amber-300" aria-hidden />
   480→                  <span className="min-w-6 font-pixel text-base tabular-nums">{stats.score}</span>
   481→                </Pill>
   482→                {/* Полоска нормы: наполняемость стаканчика */}
   483→                <Pill className="w-36 sm:w-40">
   484→                  <Popcorn className="h-4 w-4 shrink-0 text-amber-300" aria-hidden />
   485→                  <span className="tabular-nums">
   486→                    {stats.collected}/{stats.quota}
   487→                  </span>
   488→                  <span className="relative ml-auto h-2.5 w-12 overflow-hidden rounded-full bg-white/15 sm:w-14">
   489→                    <span
   490→                      className="absolute inset-y-0 left-0 rounded-full bg-amber-300 transition-[width] duration-200"
   491→                      style={{ width: `${quotaPct}%` }}
   492→                    />
   493→                  </span>
   494→                </Pill>
   495→                {/* Вместимость лапки: 3 → 4 → 5 зёрен за раз */}
   496→                <Pill>
   497→                  <span aria-hidden>🐾</span>
   498→                  <span className="tabular-nums">
   499→                    {stats.carrying}/{stats.carryMax}
   500→                  </span>
   501→                </Pill>
   502→                <Pill className={`relative overflow-hidden ${urgent ? 'animate-pulse border-red-400/60' : ''}`}>
   503→                  <span
   504→                    aria-hidden
   505→                    className="absolute inset-y-0 left-0 bg-red-400/25"
   506→                    style={{
   507→                      width: `${Math.max(0, Math.min(100, (stats.timeLeft / Math.max(1, stats.duration)) * 100))}%`,
   508→                    }}
   509→                  />
   510→                  <Timer
   511→                    className={`relative h-4 w-4 ${urgent ? 'text-red-400' : 'text-amber-300'}`}
   512→                    aria-hidden
   513→                  />

   530→                {activePerk && (
