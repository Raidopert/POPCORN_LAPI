    39→/* ---------- localStorage как внешний стор (без каскадных рендеров) ---------- */
    40→
    41→function subscribeStore(onChange: () => void): () => void {
    42→  window.addEventListener('storage', onChange)
    43→  window.addEventListener(STORE_EVENT, onChange)
    44→  return () => {
    45→    window.removeEventListener('storage', onChange)
    46→    window.removeEventListener(STORE_EVENT, onChange)
    47→  }
    48→}
    49→
    50→function useStoredValue(key: string, fallback: string): [string, (v: string) => void] {
    51→  const value = useSyncExternalStore(
    52→    subscribeStore,
    53→    () => window.localStorage.getItem(key) ?? fallback,
    54→    () => fallback,
    55→  )
    56→  const setValue = useCallback(
    57→    (v: string) => {
    58→      window.localStorage.setItem(key, v)
    59→      window.dispatchEvent(new Event(STORE_EVENT))
    60→    },
    61→    [key],
    62→  )
    63→  return [value, setValue]
    64→}

   162→/** Пиксельный сундучок (закрытый/открытый), отрисованный на canvas */
   163→function ChestImg({ open }: { open: boolean }) {
   164→  const ref = useRef<HTMLCanvasElement | null>(null)
   165→  useEffect(() => {
   166→    const c = ref.current
   167→    if (!c) return
   168→    const spr = makeChestSprite(open)
   169→    const px = 6
   170→    c.width = spr.w * px
   171→    c.height = spr.h * px
   172→    const ctx = c.getContext('2d')
   173→    if (!ctx) return
   174→    ctx.imageSmoothingEnabled = false
   175→    ctx.clearRect(0, 0, c.width, c.height)
   176→    ctx.drawImage(spr.canvas, 0, 0, c.width, c.height)
   177→  }, [open])
   178→  return <canvas ref={ref} aria-hidden className="pixelated h-auto w-16 sm:w-20" />
   179→}

   183→export function PopcornGame() {
   184→  const mainRef = useRef<HTMLElement | null>(null)
   185→  const canvasRef = useRef<HTMLCanvasElement | null>(null)
   186→  const engineRef = useRef<PopcornEngine | null>(null)
   187→  const mutedRef = useRef(false)
   188→  const transformRef = useRef<ScreenTransform>('none')
   189→  const layoutRef = useRef<() => void>(() => {})
   190→  const devRef = useRef<{ fast: boolean; forced: PerkId | null }>({ fast: false, forced: null })
   191→
   192→  const [phase, setPhase] = useState<Phase>('start')
   193→  const [stats, setStats] = useState<EngineStats>({
   194→    score: 0,
   195→    timeLeft: 60,
   196→    duration: 60,
   197→    level: 1,
   198→    quota: 10,
   199→    collected: 0,
   200→    carrying: 0,
   201→    paused: false,
   202→  })
   203→  const [bestRaw, storeBest] = useStoredValue(BEST_KEY, '0')
   204→  const [mutedRaw, storeMuted] = useStoredValue(MUTE_KEY, '0')
   205→  const best = Number(bestRaw) || 0
   206→  const muted = mutedRaw === '1'
   207→  const [result, setResult] = useState<ResultState | null>(null)
   208→  const [clearInfo, setClearInfo] = useState<ClearState | null>(null)
   209→  const [offers, setOffers] = useState<PerkDef[]>([])
   210→  const [openedIdx, setOpenedIdx] = useState<number | null>(null)
   211→  const [runLevel, setRunLevel] = useState(2)
   212→  const [perk, setPerk] = useState<PerkId | null>(null)
   213→
   214→  /* Скрытые dev-параметры URL: ?fast=1 (короткая норма) и ?perk=rot180 */
   215→  useEffect(() => {
   216→    const sp = new URLSearchParams(window.location.search)
   217→    devRef.current = {
   218→      fast: sp.has('fast'),
   219→      forced: (sp.get('perk') as PerkId | null) ?? null,
   220→    }
   221→  }, [])
   222→
   223→  /* Синхронизируем состояние звука с аудио-движком (внешняя система) */
   224→  useEffect(() => {
   225→    mutedRef.current = muted
   226→    audio.setMuted(muted)
   227→  }, [muted])
   228→
   229→  /* Раскладка холста с учётом трансформации перка (ротация/зеркало) */
   230→  const applyLayout = useCallback(() => {
   231→    const main = mainRef.current
   232→    const canvas = canvasRef.current
   233→    if (!main || !canvas) return
   234→    const cw = main.clientWidth
   235→    const ch = main.clientHeight
   236→    if (cw === 0 || ch === 0) return
   237→    const t = transformRef.current
   238→    const rot90 = t === 'rot90'
   239→    const gw = rot90 ? ch : cw
   240→    const gh = rot90 ? cw : ch
   241→    engineRef.current?.handleResize(gw, gh)
   242→    canvas.style.width = `${gw}px`
   243→    canvas.style.height = `${gh}px`
   244→    if (rot90) {
   245→      canvas.style.left = '50%'
   246→      canvas.style.top = '50%'
   247→      canvas.style.transform = 'translate(-50%, -50%) rotate(90deg)'
   248→    } else {
   249→      canvas.style.left = '0'
   250→      canvas.style.top = '0'
   251→      canvas.style.transform =
   252→        t === 'rot180' ? 'rotate(180deg)' : t === 'mirror' ? 'scaleX(-1)' : 'none'
   253→    }
   254→  }, [])
   255→
   256→  useEffect(() => {
   257→    layoutRef.current = applyLayout
   258→  }, [applyLayout])
   259→
   260→  /* Инициализация движка */
   261→  useEffect(() => {
   262→    const canvas = canvasRef.current
   263→    if (!canvas) return
   264→
   265→    const engine = new PopcornEngine(canvas, {
   266→      onStats: (s) => setStats(s),
   267→      onLevelComplete: (r) => {
   268→        setClearInfo({ level: r.level, bonus: r.bonus })
   269→        setOffers(rollPerkOffer(3, devRef.current.forced ?? undefined))
   270→        setOpenedIdx(null)
   271→        setRunLevel(r.level + 1)
   272→        setPhase('chest')
   273→      },
   274→      onEnd: (r) => {
   275→        const prevBest = Number(window.localStorage.getItem(BEST_KEY) ?? 0)
   276→        const record = r.score > prevBest
   277→        if (record) {
   278→          storeBest(String(r.score))
   279→          window.setTimeout(() => audio.play('record'), 700)
   280→        }
   281→        setResult({ score: r.score, level: r.level, record })
   282→        setPhase('ended')
   283→      },
   284→    })
   285→    engineRef.current = engine
   286→    ;(window as unknown as { __popcornEngine?: PopcornEngine }).__popcornEngine = engine
   287→    layoutRef.current()
   288→
   289→    return () => {
   290→      engine.destroy()
   291→      engineRef.current = null
   292→      delete (window as unknown as { __popcornEngine?: PopcornEngine }).__popcornEngine
   293→    }
   294→  }, [storeBest])
   295→
   296→  /* Следим за размером контейнера */
   297→  useEffect(() => {
   298→    const main = mainRef.current
   299→    if (!main || typeof ResizeObserver === 'undefined') return
   300→    const ro = new ResizeObserver(() => layoutRef.current())
   301→    ro.observe(main)
   302→    return () => ro.disconnect()
   303→  }, [])
   304→
   305→  /* Ввод: инверсия управления для перков-трансформаций.
   306→     rot180/mirror — сырые координаты: CSS-зеркало переворачивает картинку,
   307→     и лапка на экране идёт в противоположную сторону от пальца.
   308→     rot90 — мир повёрнут с заменой измерений, поэтому даём развёрнутую инверсию. */
   309→  useEffect(() => {
   310→    const main = mainRef.current
   311→    if (!main) return
   312→    const onPointer = (e: PointerEvent) => {
   313→      const eng = engineRef.current
   314→      if (!eng) return
   315→      const r = main.getBoundingClientRect()
   316→      const sx = e.clientX - r.left
   317→      const sy = e.clientY - r.top
   318→      let x = sx
   319→      let y = sy
   320→      if (transformRef.current === 'rot90') {
   321→        x = r.height - sy
   322→        y = sx
   323→      }
   324→      eng.setPointer(x, y, e.pointerType === 'touch' ? 64 : 0)
   325→    }
   326→    main.addEventListener('pointerdown', onPointer)
   327→    main.addEventListener('pointermove', onPointer)
   328→    return () => {
   329→      main.removeEventListener('pointerdown', onPointer)
   330→      main.removeEventListener('pointermove', onPointer)
   331→    }
   332→  }, [])
   333→
   334→  /* Трансформации экрана и фильтры действуют только во время уровня */
   335→  useEffect(() => {
   336→    const onLevel = phase === 'playing' || phase === 'paused'
   337→    transformRef.current = onLevel ? transformFor(perk) : 'none'
   338→    const canvas = canvasRef.current
   339→    if (canvas) {
   340→      canvas.style.filter = onLevel && perk === 'grayscale' ? 'grayscale(1) contrast(1.05)' : ''
   341→    }
   342→    layoutRef.current()
   343→  }, [perk, phase])
   344→
   345→  /* Автопауза при сворачивании вкладки: игру и музыку ставим на паузу,
   346→     чтобы троттлинг таймеров не копил отставание аудио-планировщика */
   347→  useEffect(() => {
   348→    const onVisibility = () => {
   349→      if (document.hidden) {
   350→        audio.stopMusic()
   351→        engineRef.current?.setPaused(true)
   352→        setPhase((p) => (p === 'playing' ? 'paused' : p))
   353→      } else if (engineRef.current?.phase === 'playing') {
   354→        audio.startMusic()
   355→      }
   356→    }
   357→    document.addEventListener('visibilitychange', onVisibility)
   358→    return () => document.removeEventListener('visibilitychange', onVisibility)
   359→  }, [])
   360→
   361→  const startGame = useCallback(() => {
   362→    setResult(null)
   363→    setClearInfo(null)
   364→    setPerk(null)
   365→    transformRef.current = 'none'
   366→    const canvas = canvasRef.current
   367→    if (canvas) canvas.style.filter = ''
   368→    audio.ensure()
   369→    audio.setMuted(mutedRef.current)
   370→    audio.startMusic()
   371→    audio.play('click')
   372→    engineRef.current?.startRun({ fast: devRef.current.fast })
   373→    setPhase('playing')
   374→  }, [])
   375→
   376→  const beginLevel = useCallback((lvl: number, perkId: PerkId | null) => {
   377→    audio.ensure()
   378→    audio.setMuted(mutedRef.current)
   379→    audio.startMusic()
   380→    audio.play('click')
   381→    transformRef.current = transformFor(perkId)
   382→    const canvas = canvasRef.current
   383→    if (canvas) {
   384→      canvas.style.filter = perkId === 'grayscale' ? 'grayscale(1) contrast(1.05)' : ''
   385→    }
   386→    layoutRef.current()
   387→    engineRef.current?.startLevel(lvl, perkId)
   388→    setPerk(perkId)
   389→    setPhase('playing')
   390→  }, [])
   391→
   392→  const resumeGame = useCallback(() => {
   393→    audio.play('click')
   394→    audio.startMusic()
   395→    engineRef.current?.setPaused(false)
   396→    setPhase('playing')
   397→  }, [])
   398→
   399→  const pauseGame = useCallback(() => {
   400→    audio.play('click')
   401→    audio.stopMusic()
   402→    engineRef.current?.setPaused(true)
   403→    setPhase('paused')
   404→  }, [])
   405→
   406→  const toggleMute = useCallback(() => {
   407→    storeMuted(muted ? '0' : '1')
   408→  }, [muted, storeMuted])
   409→
   410→  const openChest = useCallback((i: number) => {
   411→    audio.play('chest')
   412→    setOpenedIdx(i)
   413→    window.setTimeout(() => audio.play('perk'), 380)
   414→  }, [])
   415→
   416→  const sec = Math.ceil(stats.timeLeft)
   417→  const urgent = sec <= 10
   418→  const playing = phase === 'playing' || phase === 'paused'
   419→  const quotaPct = Math.min(100, (stats.collected / Math.max(1, stats.quota)) * 100)
   420→  const chosen: PerkDef | null = openedIdx !== null ? (offers[openedIdx] ?? null) : null
   421→  const activePerk = perk ? PERK_MAP[perk] : null
   422→
   423→  return (
   424→    <div className="flex h-dvh flex-col overflow-hidden bg-[#1d110b] select-none">
   425→      <main
   426→        ref={mainRef}
   427→        className="relative min-h-0 flex-1 touch-none overflow-hidden"
   428→        aria-label="Игровое поле: соберите рассыпанный попкорн в стаканчик лапкой куницы"
   429→      >
   430→        {/* Фон — ретро комната */}
   431→        <div
   432→          aria-hidden
   433→          className="absolute inset-0 scale-105 bg-[url('/popcorn-room.png')] bg-cover bg-center"
   434→        />
   435→        <div
   436→          aria-hidden
   437→          className="absolute inset-0 bg-gradient-to-b from-black/50 via-black/10 to-black/55"
   438→        />
   439→
   440→        {/* Игровой холст (трансформации перков применяются инлайн) */}
   441→        <canvas
   442→          ref={canvasRef}
   443→          className="absolute z-10 touch-none"
   444→          onContextMenu={(e) => e.preventDefault()}
   445→        />
   446→
   447→        {/* HUD */}
   448→        <div className="pointer-events-none absolute inset-x-0 top-0 z-20 flex items-start justify-between gap-2 p-3 pt-[max(0.75rem,env(safe-area-inset-top))]">
   449→          <div className="flex flex-col gap-2">
   450→            {playing ? (
   451→              <>
   452→                <Pill>
   453→                  <Popcorn className="h-4 w-4 text-amber-300" aria-hidden />
   454→                  <span className="min-w-6 font-pixel text-base tabular-nums">{stats.score}</span>
   455→                </Pill>
   456→                {/* Полоска нормы: наполняемость стаканчика */}
   457→                <Pill className="w-36 sm:w-40">
   458→                  <Popcorn className="h-4 w-4 shrink-0 text-amber-300" aria-hidden />
   459→                  <span className="tabular-nums">
   460→                    {stats.collected}/{stats.quota}
   461→                  </span>
   462→                  <span className="relative ml-auto h-2.5 w-12 overflow-hidden rounded-full bg-white/15 sm:w-14">
   463→                    <span
   464→                      className="absolute inset-y-0 left-0 rounded-full bg-amber-300 transition-[width] duration-200"
   465→                      style={{ width: `${quotaPct}%` }}
   466→                    />
   467→                  </span>
   468→                </Pill>
   469→                <Pill className={`relative overflow-hidden ${urgent ? 'animate-pulse border-red-400/60' : ''}`}>
   470→                  <span
   471→                    aria-hidden
   472→                    className="absolute inset-y-0 left-0 bg-red-400/25"
   473→                    style={{
   474→                      width: `${Math.max(0, Math.min(100, (stats.timeLeft / Math.max(1, stats.duration)) * 100))}%`,
   475→                    }}
   476→                  />
   477→                  <Timer
   478→                    className={`relative h-4 w-4 ${urgent ? 'text-red-400' : 'text-amber-300'}`}
   479→                    aria-hidden
   480→                  />
   481→                  <span className="relative tabular-nums">{sec} с</span>
   482→                </Pill>
   483→              </>
   484→            ) : (
   485→              <Pill>
   486→                <Trophy className="h-4 w-4 text-amber-300" aria-hidden />
   487→                <span className="font-pixel text-base tabular-nums">{best}</span>
   488→              </Pill>
   489→            )}
   490→          </div>
   491→          <div className="flex flex-col items-end gap-2">
   492→            {playing && (
   493→              <>
   494→                <Pill>
   495→                  <span className="font-pixel text-sm">УР. {stats.level}</span>
   496→                </Pill>
   497→                {activePerk && (
   498→                  <Pill
   499→                    className={
   500→                      activePerk.curse
   501→                        ? 'border-red-400/50 text-red-200'
   502→                        : 'border-emerald-300/50 text-emerald-200'
   503→                    }
   504→                  >
   505→                    <span aria-hidden>{activePerk.icon}</span>
   506→                    <span className="max-w-36 truncate text-[11px] sm:max-w-none sm:text-xs">
   507→                      {activePerk.name}
   508→                    </span>
   509→                  </Pill>
   510→                )}
   511→              </>
   512→            )}
   513→            <div className="flex gap-2">
   514→              {phase === 'playing' && (
   515→                <IconBtn onClick={pauseGame} label="Пауза">
   516→                  <Pause className="h-5 w-5" aria-hidden />
   517→                </IconBtn>
   518→              )}
   519→              <IconBtn onClick={toggleMute} label={muted ? 'Включить звук' : 'Выключить звук'}>
   520→                {muted ? <VolumeX className="h-5 w-5" aria-hidden /> : <Volume2 className="h-5 w-5" aria-hidden />}
   521→              </IconBtn>
   522→            </div>
   523→          </div>
   524→        </div>
   525→
   526→        {/* Оверлеи */}
   527→        <AnimatePresence>
   528→          {phase === 'start' && (
   529→            <motion.div
   530→              key="start"
   531→              className="absolute inset-0 z-30 flex items-center justify-center overflow-y-auto bg-black/60 p-4 backdrop-blur-sm"
   532→              initial={{ opacity: 0 }}
   533→              animate={{ opacity: 1 }}
   534→              exit={{ opacity: 0 }}
   535→            >
   536→              <motion.div
   537→                className="relative my-auto w-full max-w-sm rounded-2xl border-4 border-[#8a5330] bg-[#FFF6E6] p-4 text-center shadow-[0_18px_50px_rgba(0,0,0,0.55)] [@media(max-height:700px)]:p-3 sm:p-6"
