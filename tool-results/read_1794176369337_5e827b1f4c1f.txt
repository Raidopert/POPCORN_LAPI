     1→/**
     2→ * Игровой движок «Кошачий попкорнный Беспредел» — рогалик-режим.
     3→ * Canvas + requestAnimationFrame, независим от React.
     4→ *
     5→ * Игра бесконечна: каждый уровень = 60 секунд и норма (квота) попкорна.
     6→ * Выполнил норму → уровень пройден, сундучок с перком, дальше сложнее.
     7→ * Не успел → конец забега. Перки меняют правила одного уровня.
     8→ */

   117→export class PopcornEngine {
   118→  private canvas: HTMLCanvasElement
   119→  private ctx: CanvasRenderingContext2D
   120→  private cb: EngineCallbacks
   121→  private raf = 0
   122→  private lastTs = 0
   123→  private destroyed = false
   124→
   125→  private W = 0
   126→  private H = 0
   127→  private dpr = 1
   128→  private u = 1
   129→
   130→  phase: 'idle' | 'playing' | 'clear' | 'ended' = 'idle'
   131→  paused = false
   132→
   133→  private timeLeft = 0
   134→  private duration = LEVEL_SECONDS
   135→  private score = 0
   136→  private lvl = 1
   137→  private perk: PerkId | null = null
   138→  private quota = 10
   139→  private collected = 0
   140→  private wave = 1
   141→  private waveSize = 7
   142→  private cornChance = 0.12
   143→  private goldenChance = 0.12
   144→  private baseWaveDelay = 0.82
   145→  private waveDelay = -1
   146→  private pendingClear = -1
   147→  private debugFast = false
   148→
   149→  private pieces: FieldPiece[] = []
   150→  private carried: Carried[] = []
   151→  private flying: Flying[] = []
   152→  private particles: Particle[] = []
   153→  private texts: FloatText[] = []
   154→  private paw = { x: 0, y: 0, tx: 0, ty: 0, seen: false, pulse: 0, hurt: 0 }
   155→  private cup = { x: 0, y: 0, squash: 0 }
   156→  private banner = { text: '', t: 99 }
   157→  private lastSec = -1
   158→  private audioStage = -1
   159→  private lastStatsKey = ''
   160→
   161→  private cupScale = 5
   162→  private cupH = 0

   164→  private sp: {
   165→    popcorn: Sprite
   166→    golden: Sprite
   167→    corn: Sprite
   168→    cup: Sprite

   173→  constructor(canvas: HTMLCanvasElement, cb: EngineCallbacks = {}) {
   174→    this.canvas = canvas
   175→    this.ctx = canvas.getContext('2d')!
   176→    this.cb = cb
   177→    this.sp = {
   178→      popcorn: makePopcornSprite(false),
   179→      golden: makePopcornSprite(true),
   180→      corn: makeCornSprite(),
   181→      cup: makeCupSprite(),
   182→      paw: makePawSprite(),
   183→      sparkle: makeSparkleSprite(),
   184→    }
   185→    this.lastTs = performance.now()
   186→    this.raf = requestAnimationFrame(this.loop)
   187→  }
   188→
   189→  destroy(): void {
   190→    this.destroyed = true
   191→    cancelAnimationFrame(this.raf)
   192→  }

   194→  /* ================= публичное API ================= */
   195→
   196→  /** Новый забег: счёт с нуля, уровень 1 без перка */
   197→  startRun(opts: { fast?: boolean } = {}): void {
   198→    this.debugFast = !!opts.fast
   199→    this.score = 0
   200→    this.startLevel(1, null)
   201→  }
   202→
   203→  /** Следующий уровень забега (счёт сохраняется) */
   204→  startLevel(level: number, perk: PerkId | null): void {
   205→    this.lvl = level
   206→    this.perk = perk
   207→    this.duration = perk === 'timeplus' ? LEVEL_SECONDS_PERK : LEVEL_SECONDS
   208→    this.quota = this.debugFast ? 3 : quotaFor(level, perk === 'doublepoints')
   209→    this.collected = 0
   210→    this.wave = 1
   211→    this.waveSize = Math.min(16, 7 + Math.floor(level * 0.8))
   212→    this.cornChance = Math.min(0.3, 0.12 + (level - 1) * 0.02)
   213→    if (perk === 'cornrain') this.cornChance = Math.min(0.5, this.cornChance * 2)
   214→    this.goldenChance = perk === 'goldenvein' ? 0.26 : 0.12
   215→    this.baseWaveDelay = Math.max(0.35, 0.82 - (level - 1) * 0.04)
   216→
   217→    this.phase = 'playing'
   218→    this.paused = false
   219→    this.timeLeft = this.duration
   220→    this.pendingClear = -1
   221→    this.pieces = []
   222→    this.carried = []
   223→    this.flying = []
   224→    this.particles = []
   225→    this.texts = []
   226→    this.waveDelay = -1
   227→    this.lastSec = Math.ceil(this.timeLeft)
   228→    this.banner = { text: `УРОВЕНЬ ${level}`, t: 0 }
   229→    audio.setIntensity(0)
   230→    audio.setMusicLevel(level >= 3 ? 1 : 0)
   231→    this.audioStage = 0
   232→    if (!this.paw.seen) {
   233→      this.paw.x = this.cup.x
   234→      this.paw.y = this.H * 0.4
   235→      this.paw.tx = this.paw.x
   236→      this.paw.ty = this.paw.y
   237→    }
   238→    this.paw.hurt = 0
   239→    audio.play('wave')
   240→    this.spawnWave()
   241→    this.emitStats(true)
   242→  }
   243→
   244→  setPaused(p: boolean): void {
   245→    if (this.phase !== 'playing') return
   246→    this.paused = p
   247→    this.emitStats(true)
   248→  }
   249→
   250→  /** Координаты в системе игрового поля (трансформации экрана считает компонент) */
   251→  setPointer(x: number, y: number, lift = 0): void {
   252→    if (this.phase !== 'playing' || this.paused) return
   253→    this.paw.tx = x
   254→    this.paw.ty = y - lift * this.u
   255→    this.paw.seen = true
   256→  }
   257→
   258→  /* ================= размеры ================= */
   259→
   260→  /** Размер игрового поля в CSS-пикселях (задаёт компонент с учётом перков) */
   261→  handleResize(w: number, h: number): void {
   262→    if (w === 0 || h === 0) return
   263→
   264→    const oldW = this.W
   265→    const oldH = this.H
   266→    this.dpr = Math.min(window.devicePixelRatio || 1, 2)
   267→    this.W = w
   268→    this.H = h
   269→    this.canvas.width = Math.round(w * this.dpr)
   270→    this.canvas.height = Math.round(h * this.dpr)
   271→    this.ctx.setTransform(this.dpr, 0, 0, this.dpr, 0, 0)
   272→    this.ctx.imageSmoothingEnabled = false
   273→
   274→    this.u = Math.max(0.7, Math.min(1.5, Math.min(w, h) / 420))
   275→
   276→    if (oldW > 0 && oldH > 0 && (oldW !== w || oldH !== h)) {
   277→      const kx = w / oldW
   278→      const ky = h / oldH
   279→      for (const p of this.pieces) {
   280→        p.x *= kx
   281→        p.y *= ky
   282→      }
   283→      this.paw.x *= kx
   284→      this.paw.y *= ky
   285→      this.paw.tx *= kx
   286→      this.paw.ty *= ky
   287→    }
   288→
   289→    this.cupScale = 4.6 * this.u
   290→    this.cupH = 26 * this.cupScale
   291→    this.cup.x = w / 2
   292→    this.cup.y = h - this.cupH / 2 - 12 * this.u
   293→  }
   294→
   295→  /* ================= волны ================= */
   296→
   297→  private spawnWave(): void {
   298→    let cornOnField = this.pieces.filter((p) => p.kind === 'corn').length
   299→    for (let i = 0; i < this.waveSize; i++) {
   300→      const kind = this.decideKind(cornOnField)
   301→      if (kind === 'corn') cornOnField++
   302→      this.spawnPiece(i, kind)
   303→    }
   304→  }
   305→
   306→  private decideKind(cornOnField: number): PieceKind {
   307→    const r = Math.random()
   308→    if (r < this.cornChance && cornOnField < CORN_MAX) return 'corn'
   309→    return Math.random() < this.goldenChance ? 'golden' : 'popcorn'
   310→  }
   311→
   312→  private spawnPiece(index: number, kind: PieceKind): void {
   313→    const u = this.u
   314→    const mX = 46 * u
   315→    const top = 80 + 16 * u
   316→    const bottom = this.H - 46 * u
   317→    let x = this.W / 2
   318→    let y = this.H / 2
   319→    for (let tries = 0; tries < 40; tries++) {
   320→      x = mX + Math.random() * (this.W - mX * 2)
   321→      y = top + Math.random() * Math.max(10, bottom - top)
   322→      const dc = Math.hypot(x - this.cup.x, y - this.cup.y)
   323→      if (dc <= 135 * u) continue
   324→      // не даём зёрнам застрять в кластере — маршрут всегда проходим
   325→      let crowded = false
   326→      for (const q of this.pieces) {
   327→        if (Math.hypot(x - q.x, y - q.y) < 40 * u) {
   328→          crowded = true
   329→          break
   330→        }
   331→      }
   332→      if (!crowded) break
   333→    }
   334→    this.pieces.push({
   335→      x,
   336→      y,
   337→      rot: (Math.random() - 0.5) * 0.7,
   338→      kind,
   339→      phase: Math.random() * Math.PI * 2,
   340→      spawn: SPAWN_DUR + index * 0.045,
   341→      dA: Math.random() * Math.PI * 2,
   342→    })
   343→  }
   344→
   345→  /* ================= перки-механики ================= */
   346→
   347→  /** «Скользкий пол»: зёрна катаются по полу */
   348→  private driftPieces(dt: number): void {
   349→    const u = this.u
   350→    const sp = 17 * u
   351→    const mX = 34 * u
   352→    const top = 90
   353→    const bottom = this.H - 42 * u
   354→    for (const p of this.pieces) {
   355→      p.dA += (Math.random() - 0.5) * dt * 2.4
   356→      p.x += Math.cos(p.dA) * sp * dt
   357→      p.y += Math.sin(p.dA) * sp * dt
   358→      if (p.x < mX) {
   359→        p.x = mX
   360→        p.dA = Math.PI - p.dA
   361→      }
   362→      if (p.x > this.W - mX) {
   363→        p.x = this.W - mX
   364→        p.dA = Math.PI - p.dA
   365→      }
   366→      if (p.y < top) {
   367→        p.y = top
   368→        p.dA = -p.dA
   369→      }
   370→      if (p.y > bottom) {
   371→        p.y = bottom
   372→        p.dA = -p.dA
   373→      }
   374→    }
   375→  }
   376→
   377→  /** «Магнитная лапа»: попкорн сам тянется к лапке */
   378→  private magnetPull(dt: number): void {
   379→    const u = this.u
   380→    const r = 115 * u
   381→    const k = Math.min(1, dt * 3.4)
   382→    for (const p of this.pieces) {
   383→      if (p.kind === 'corn' || p.spawn > 0.12) continue
   384→      const d = Math.hypot(this.paw.x - p.x, this.paw.y - p.y)
   385→      if (d < r && d > 1) {
   386→        p.x += (this.paw.x - p.x) * k
   387→        p.y += (this.paw.y - p.y) * k
   388→      }
   389→    }
   390→  }
   391→
   392→  /* ================= обновление ================= */
   393→
   394→  private update(dt: number): void {
   395→    const u = this.u
   396→
   397→    // Плавное следование лапки за пальцем/курсором («Тяжёлая лапа» — вдвое медленнее)
   398→    if (!this.paw.seen) {
   399→      this.paw.tx = this.cup.x + Math.sin(performance.now() * 0.001) * 80 * u
   400→      this.paw.ty = this.H * 0.35
   401→    }
   402→    const k = 1 - Math.exp(-dt * (this.perk === 'slowpaw' ? 6.5 : 14))
   403→    this.paw.x += (this.paw.tx - this.paw.x) * k
   404→    this.paw.y += (this.paw.ty - this.paw.y) * k
   405→    this.paw.x = Math.max(22 * u, Math.min(this.W - 22 * u, this.paw.x))
   406→    this.paw.y = Math.max(70, Math.min(this.H - 20 * u, this.paw.y))
   407→
   408→    if (this.perk === 'drift') this.driftPieces(dt)
   409→    if (this.perk === 'magnet') this.magnetPull(dt)
   410→
   411→    // Радиус подбора зависит от перка лапки
   412→    const grabR = (this.perk === 'bigpaw' ? 58 : this.perk === 'smallpaw' ? 34 : 44) * u
   413→
   414→    // Подбор попкорна / ловушка-кукуруза
   415→    for (let i = this.pieces.length - 1; i >= 0; i--) {
   416→      const p = this.pieces[i]
   417→      if (p.spawn > 0.12) continue
   418→      const d = Math.hypot(this.paw.x - p.x, this.paw.y - p.y)
   419→      if (p.kind === 'corn') {
   420→        // Кукуруза — мина: задел = минус балл
   421→        if (d < 38 * u) {
   422→          this.pieces.splice(i, 1)
   423→          this.score = Math.max(0, this.score - 1)
   424→          this.paw.hurt = 1
   425→          this.texts.push({
   426→            x: this.paw.x,
   427→            y: this.paw.y - 34 * u,
   428→            text: '-1',
   429→            color: '#FF6B5E',
   430→            t: 0,
   431→            dur: 0.9,
   432→            size: 24 * u,
   433→          })
   434→          this.crumbs(p.x, p.y, '#E8960C', 7)
   435→          audio.play('penalty')
   436→        }
   437→        continue
   438→      }
   439→      if (d < grabR && this.carried.length < CARRY_MAX) {
   440→        this.pieces.splice(i, 1)
   441→        this.carried.push({ golden: p.kind === 'golden', grabT: 0 })
   442→        this.paw.pulse = 1
   443→        this.crumbs(p.x, p.y, p.kind === 'golden' ? '#FFE066' : '#FFF3D6', 6)
   444→        audio.play(p.kind === 'golden' ? 'golden' : 'pickup')
   445→      }
   446→    }
   447→
   448→    // Высыпание в стаканчик
   449→    if (this.carried.length > 0) {
   450→      const dc = Math.hypot(this.paw.x - this.cup.x, this.paw.y - this.cup.y)
   451→      if (dc < 78 * u) {
   452→        let base = 0
   453→        let hasGold = false
   454→        for (const c of this.carried) {
   455→          base += c.golden ? 5 : 1
   456→          hasGold = hasGold || c.golden
   457→        }
   458→        const mult = this.perk === 'doublepoints' ? 2 : 1
   459→        const sum = base * mult
   460→        this.collected += this.carried.length
   461→        this.carried.forEach((c, i) => {
   462→          this.flying.push({
   463→            x0: this.paw.x + (i - 1) * 8 * u,
   464→            y0: this.paw.y - 20 * u - i * 10 * u,
   465→            golden: c.golden,
   466→            t: 0,
   467→          })
   468→        })
   469→        this.carried = []
   470→        this.score += sum
   471→        this.cup.squash = 1
   472→        this.texts.push({
   473→          x: this.cup.x,
   474→          y: this.cup.y - this.cupH * 0.8,
   475→          text: `+${sum}`,
   476→          color: hasGold ? '#FFC93C' : '#FFFFFF',
   477→          t: 0,
   478→          dur: 0.9,
   479→          size: 26 * u,
   480→        })
   481→        audio.play('deposit')
   482→      }
   483→    }
   484→
   485→    // Летящие в стаканчик зёрна
   486→    for (let i = this.flying.length - 1; i >= 0; i--) {
   487→      const f = this.flying[i]
   488→      f.t += dt / 0.3
   489→      if (f.t >= 1) {
   490→        this.flying.splice(i, 1)
   491→        this.crumbs(this.cup.x, this.cup.y - this.cupH * 0.5, '#FFF6EC', 3)
   492→      }
   493→    }
   494→
   495→    // Вся волна собрана → следующая. Кукуруза-мины НЕ мешает: считаем только попкорн
   496→    const hasPopcorn = this.pieces.some((p) => p.kind !== 'corn')
   497→    if (
   498→      this.waveDelay < 0 &&
   499→      !hasPopcorn &&
   500→      this.carried.length === 0 &&
   501→      this.flying.length === 0
   502→    ) {
   503→      this.waveDelay = this.baseWaveDelay
   504→    }
   505→    if (this.waveDelay >= 0) {
   506→      this.waveDelay -= dt
   507→      if (this.waveDelay <= 0) {
   508→        this.waveDelay = -1
   509→        this.wave++
   510→        this.banner = { text: `ВОЛНА ${this.wave}`, t: 0 }
   511→        audio.play('wave')
   512→        this.spawnWave()
   513→      }
   514→    }
   515→
   516→    // Норма выполнена → короткая пауза (зёрна долетают) → уровень пройден
   517→    if (this.pendingClear < 0 && this.collected >= this.quota) {
   518→      this.pendingClear = CLEAR_DELAY
   519→    }
   520→    if (this.pendingClear >= 0) {
   521→      this.pendingClear -= dt
   522→      if (this.pendingClear <= 0) {
   523→        this.levelComplete()
   524→        return
   525→      }
   526→    }
   527→
   528→    // Музыка зависит от оставшегося времени: >30с база, ≤30с +25%, ≤15с +40%, ≤5с 240 BPM
   529→    const stage = this.timeLeft > 30 ? 0 : this.timeLeft > 15 ? 1 : this.timeLeft > 5 ? 2 : 3
   530→    if (stage !== this.audioStage) {
   531→      this.audioStage = stage
   532→      audio.setIntensity(stage)
   533→    }
   534→
   535→    // Таймер
   536→    this.timeLeft -= dt
   537→    const sec = Math.max(0, Math.ceil(this.timeLeft))
   538→    if (sec !== this.lastSec) {
   539→      this.lastSec = sec
   540→      if (sec <= 5 && sec > 0) audio.play('tick')
   541→    }
   542→    if (this.timeLeft <= 0) {
   543→      this.timeLeft = 0
   544→      this.end()
   545→      return
   546→    }
   547→
   548→    // Косметика
   549→    this.paw.pulse = Math.max(0, this.paw.pulse - dt * 5)
   550→    this.paw.hurt = Math.max(0, this.paw.hurt - dt * 2.2)
   551→    this.cup.squash = Math.max(0, this.cup.squash - dt * 4)
   552→    this.banner.t += dt
   553→    for (const p of this.pieces) p.spawn = Math.max(0, p.spawn - dt)
   554→    for (const c of this.carried) c.grabT = Math.min(1, c.grabT + dt * 8)
   555→    for (let i = this.particles.length - 1; i >= 0; i--) {
   556→      const p = this.particles[i]
   557→      p.t += dt
   558→      p.vy += 900 * u * dt
   559→      p.x += p.vx * dt
   560→      p.y += p.vy * dt
   561→      if (p.t >= p.ttl) this.particles.splice(i, 1)
   562→    }
   563→    for (let i = this.texts.length - 1; i >= 0; i--) {
   564→      const t = this.texts[i]
   565→      t.t += dt
   566→      if (t.t >= t.dur) this.texts.splice(i, 1)
   567→    }
   568→  }
   569→
   570→  private levelComplete(): void {
   571→    this.phase = 'clear'
   572→    const bonus = Math.ceil(this.timeLeft)
   573→    this.score += bonus
   574→    audio.stopMusic()
   575→    audio.play('levelwin')
   576→    this.emitStats(true)
   577→    this.cb.onLevelComplete?.({ level: this.lvl, score: this.score, bonus })
   578→  }
   579→
   580→  private end(): void {
   581→    this.phase = 'ended'
   582→    audio.stopMusic()
   583→    audio.play('gameover')
   584→    this.emitStats(true)
   585→    this.cb.onEnd?.({ score: this.score, level: this.lvl })
   586→  }
   587→
   588→  private crumbs(x: number, y: number, color: string, n: number): void {
   589→    const u = this.u
   590→    for (let i = 0; i < n; i++) {
   591→      const a = Math.random() * Math.PI * 2
   592→      const sp = (60 + Math.random() * 120) * u
   593→      this.particles.push({
   594→        x,
   595→        y,
   596→        vx: Math.cos(a) * sp,
   597→        vy: Math.sin(a) * sp - 60 * u,
   598→        t: 0,
   599→        ttl: 0.35 + Math.random() * 0.25,
   600→        color,
   601→        size: (2 + Math.random() * 2.5) * u,
   602→      })
   603→    }
   604→  }

   606→  /* ================= цикл ================= */
   607→
   608→  private loop = (ts: number): void => {
   609→    if (this.destroyed) return
   610→    const dt = Math.min(0.05, (ts - this.lastTs) / 1000)
   611→    this.lastTs = ts
   612→    if (this.phase === 'playing' && !this.paused) {
   613→      try {
   614→        this.update(dt)
   615→      } catch (err) {
   616→        // Сбой кадра не должен замораживать игру — логируем и продолжаем
   617→        console.error('[popcorn] update error:', err)
   618→      }
   619→    }
   620→    this.render(ts)
   621→    this.emitStats(false)
   622→    this.raf = requestAnimationFrame(this.loop)
   623→  }
   624→
   625→  private emitStats(force: boolean): void {
   626→    if (!this.cb.onStats) return
   627→    const sec = Math.ceil(this.timeLeft)
   628→    const key = `${this.score}|${this.lvl}|${this.quota}|${this.collected}|${this.carried.length}|${sec}|${this.paused}`
   629→    if (!force && key === this.lastStatsKey) return
   630→    this.lastStatsKey = key
   631→    this.cb.onStats({
   632→      score: this.score,
   633→      timeLeft: this.timeLeft,
   634→      duration: this.duration,
   635→      level: this.lvl,
   636→      quota: this.quota,
   637→      collected: this.collected,
   638→      carrying: this.carried.length,
   639→      paused: this.paused,
   640→    })
   641→  }
   642→
   643→  /* ================= отрисовка ================= */
   644→
   645→  private render(ts: number): void {
   646→    const { ctx } = this
   647→    ctx.clearRect(0, 0, this.W, this.H)
   648→    if (this.phase === 'idle') return
   649→
   650→    this.drawCup()
   651→    this.drawPieces(ts)
   652→    this.drawFlying()
   653→    this.drawParticles()
   654→    this.drawPaw(ts)
   655→    this.drawTexts()
   656→    this.drawBanner()
   657→    if (this.perk === 'fog' && this.phase === 'playing') this.drawFog()
   658→
   659→    if (this.phase === 'ended' || this.phase === 'clear') {
   660→      ctx.fillStyle = 'rgba(24, 10, 4, 0.35)'
   661→      ctx.fillRect(0, 0, this.W, this.H)
   662→    }
   663→  }
   664→
   665→  /** «Попкорнный туман»: ступенчатые кольца темноты вокруг лапки (8-битный стиль) */
   666→  private drawFog(): void {
   667→    const { ctx } = this
   668→    const u = this.u
   669→    const rings: Array<[number, number]> = [
   670→      [150, 0.86],
   671→      [120, 0.8],
   672→      [95, 0.66],
   673→      [72, 0.45],
   674→      [52, 0.22],
   675→    ]
   676→    for (const [r, a] of rings) {
   677→      ctx.fillStyle = `rgba(12, 6, 2, ${a})`
   678→      ctx.beginPath()
   679→      ctx.rect(0, 0, this.W, this.H)
   680→      ctx.arc(this.paw.x, this.paw.y, r * u, 0, Math.PI * 2)
   681→      ctx.fill('evenodd')
   682→    }
   683→  }
   684→
   685→  private drawCup(): void {
   686→    const { ctx } = this
   687→    const s = this.cupScale
   688→    const sq = this.cup.squash
   689→    const sx = 1 + 0.08 * sq
   690→    const sy = 1 - 0.1 * sq
   691→
   692→    // Тень
   693→    ctx.fillStyle = 'rgba(0, 0, 0, 0.22)'
   694→    ctx.beginPath()
   695→    ctx.ellipse(
   696→      this.cup.x,
   697→      this.cup.y + (this.cupH / 2) * sy + 5 * this.u,
   698→      48 * this.u * sx,
   699→      10 * this.u,
   700→      0,
   701→      0,
   702→      Math.PI * 2,
   703→    )
   704→    ctx.fill()
   705→
   706→    // «Шапка» попкорна растёт вместе с выполнением нормы
   707→    const progress = this.quota > 0 ? this.collected / this.quota : 0
   708→    const puffs = Math.min(PUFF_POS.length, Math.ceil(progress * PUFF_POS.length))
   709→    const mouthY = this.cup.y - 13 * s * sy
   710→    for (let i = 0; i < puffs; i++) {
   711→      const [ox, oy] = PUFF_POS[i]
   712→      drawSprite(ctx, this.sp.popcorn, this.cup.x + ox * s * 0.85, mouthY + oy * s * 0.75, s * 0.52, {
   713→        rot: i * 0.7,
   714→      })
   715→    }
   716→
   717→    // Корпус стаканчика
   718→    drawSprite(ctx, this.sp.cup, this.cup.x, this.cup.y, s, { sx, sy })
   719→  }
   720→
   721→  private drawPieces(ts: number): void {
   722→    const { ctx } = this
   723→    const u = this.u
   724→    for (const p of this.pieces) {
   725→      // Мины-зерна статичны (никаких прыжков) — попкорн слегка «дышит»
   726→      const bobAmp = p.kind === 'corn' ? 0 : 2.5
   727→      const bob = Math.sin(ts * 0.004 + p.phase) * bobAmp * u
   728→      const y = p.y + bob
   729→      const pop = p.spawn > 0 ? easeOutBack(1 - p.spawn / (SPAWN_DUR + 0.5)) : 1
   730→      const base = p.kind === 'corn' ? 2.3 : 2.6
   731→      const scale = base * u * Math.max(0.05, pop)
   732→
   733→      // Тень
   734→      ctx.fillStyle = 'rgba(0, 0, 0, 0.16)'
   735→      ctx.beginPath()
   736→      ctx.ellipse(p.x, p.y + 9 * u, 12 * u * pop, 4 * u * pop, 0, 0, Math.PI * 2)
   737→      ctx.fill()
   738→
   739→      const sprite = p.kind === 'corn' ? this.sp.corn : p.kind === 'golden' ? this.sp.golden : this.sp.popcorn
   740→      drawSprite(ctx, sprite, p.x, y, scale, { rot: p.rot })
   741→      if (p.kind === 'golden') {
   742→        drawSprite(ctx, this.sp.sparkle, p.x + 10 * u, y - 11 * u, 1.15 * u, {
   743→          rot: ts * 0.004,
   744→          alpha: 0.95,
   745→        })
   746→      }
   747→    }
   748→  }
   749→
   750→  private drawFlying(): void {
   751→    const { ctx } = this
   752→    const u = this.u
   753→    const tx = this.cup.x
   754→    const ty = this.cup.y - 10 * this.cupScale
   755→    for (const f of this.flying) {
   756→      const q = easeOutQuad(f.t)
   757→      const x = f.x0 + (tx - f.x0) * q
   758→      const y = f.y0 + (ty - f.y0) * q - Math.sin(q * Math.PI) * 30 * u
   759→      drawSprite(ctx, f.golden ? this.sp.golden : this.sp.popcorn, x, y, 2.6 * u * (1 - 0.45 * q), {
   760→        rot: q * 3,
   761→      })
   762→    }
   763→  }
   764→
   765→  private drawPaw(ts: number): void {
   766→    const { ctx } = this
   767→    const u = this.u
   768→    if (!this.paw.seen && this.phase === 'idle') return
   769→
   770→    // Тряска при штрафе + сильная дрожь перка «Нервная лапа»
   771→    const shake = Math.sin(ts * 0.055) * 5 * this.paw.hurt
   772→    const shiver = this.perk === 'shaky'
   773→    const jx = shiver ? Math.sin(ts * 0.021) * 9 * u + Math.sin(ts * 0.047) * 6 * u : 0
   774→    const jy = shiver ? Math.cos(ts * 0.017) * 7 * u + Math.sin(ts * 0.053) * 4 * u : 0
   775→    const px = this.paw.x + shake + jx
   776→    const py = this.paw.y + jy
   777→
   778→    // Тень лапки
   779→    ctx.fillStyle = 'rgba(0, 0, 0, 0.18)'
   780→    ctx.beginPath()
   781→    ctx.ellipse(px, py + 28 * u, 26 * u, 7 * u, 0, 0, Math.PI * 2)
   782→    ctx.fill()
   783→
   784→    const basePaw = this.perk === 'bigpaw' ? 4.8 : this.perk === 'smallpaw' ? 2.9 : 3.8
   785→    const s = basePaw * u * (1 + 0.13 * this.paw.pulse)
   786→    drawSprite(ctx, this.sp.paw, px, py, s)
   787→
   788→    // Зёрна в лапке (стопкой над подушечкой)
   789→    this.carried.forEach((c, i) => {
   790→      const wig = Math.sin(ts * 0.012 + i * 2) * 1.6 * u
   791→      const y = py - 32 * u - i * 15 * u + (1 - c.grabT) * 10 * u
   792→      drawSprite(ctx, c.golden ? this.sp.golden : this.sp.popcorn, px + wig, y, 2.3 * u, {
   793→        rot: Math.sin(ts * 0.006 + i) * 0.12,
   794→      })
   795→    })
   796→  }
   797→
   798→  private drawParticles(): void {
   799→    const { ctx } = this
   800→    for (const p of this.particles) {
   801→      ctx.globalAlpha = 1 - p.t / p.ttl
   802→      ctx.fillStyle = p.color
   803→      ctx.fillRect(p.x - p.size / 2, p.y - p.size / 2, p.size, p.size)
   804→    }
   805→    ctx.globalAlpha = 1
   806→  }
   807→
   808→  private drawTexts(): void {
   809→    const { ctx } = this
   810→    for (const t of this.texts) {
   811→      const q = t.t / t.dur
   812→      const alpha = q < 0.15 ? q / 0.15 : 1 - Math.max(0, (q - 0.55) / 0.45)
   813→      const y = t.y - 42 * this.u * q
   814→      ctx.save()
   815→      ctx.globalAlpha = Math.max(0, Math.min(1, alpha))
   816→      ctx.font = `600 ${t.size}px "Handjet", system-ui, sans-serif`
   817→      ctx.textAlign = 'center'
   818→      ctx.textBaseline = 'middle'
   819→      ctx.lineWidth = Math.max(3, t.size * 0.16)
   820→      ctx.strokeStyle = 'rgba(46, 18, 6, 0.9)'
   821→      ctx.strokeText(t.text, t.x, y)
   822→      ctx.fillStyle = t.color
   823→      ctx.fillText(t.text, t.x, y)
   824→      ctx.restore()
   825→    }
   826→  }
   827→
   828→  private drawBanner(): void {
   829→    if (this.banner.t > 1.5) return
   830→    const { ctx } = this
   831→    const t = this.banner.t
   832→    const alpha = t < 0.2 ? t / 0.2 : t > 1.1 ? Math.max(0, (1.5 - t) / 0.4) : 1
   833→    const pop = 1 + Math.max(0, 1 - t / 0.25) * 0.5
   834→    const size = 34 * this.u * pop
   835→    ctx.save()
   836→    ctx.globalAlpha = alpha
   837→    ctx.font = `700 ${size}px "Handjet", system-ui, sans-serif`
   838→    ctx.textAlign = 'center'
   839→    ctx.textBaseline = 'middle'
   840→    ctx.lineWidth = Math.max(4, size * 0.18)
   841→    ctx.strokeStyle = 'rgba(46, 18, 6, 0.9)'
   842→    ctx.strokeText(this.banner.text, this.W / 2, this.H * 0.3)
   843→    ctx.fillStyle = '#FFE066'
   844→    ctx.fillText(this.banner.text, this.W / 2, this.H * 0.3)
   845→    ctx.restore()
   846→  }
